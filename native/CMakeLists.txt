cmake_minimum_required(VERSION 3.16)
project(DccKdenliveBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
add_library(dcc_bridge_transport STATIC bridge_server.cpp)
target_link_libraries(dcc_bridge_transport PUBLIC Qt6::Core Qt6::Network)
target_include_directories(dcc_bridge_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_executable(bridge_tests tests/bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE dcc_bridge_transport Qt6::Test)
add_test(NAME bridge_protocol COMMAND bridge_tests -o bridge-results.xml,junitxml -o -,txt)
